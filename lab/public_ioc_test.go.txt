// Only public metadata is supplied. No malware file is fetched or executed.
func TestPublicAdvisoryHashes(t *testing.T) {
 s:=New()
 if err:=s.LoadIOC("lab/public_ioc_2026-11-25.sha256");err!=nil{t.Fatal(err)}
 if len(s.IOC)!=12{t.Fatalf("expected 12 distinct public indicators, got %d",len(s.IOC))}
 matches,negative:=0,0
 for hash,label:=range s.IOC{
  for _,value:=range []string{hash,strings.ToUpper(hash)}{
   f:=analyze("metadata-only.txt",[]byte("Benign laboratory placeholder."),value,s.IOC)
   if f==nil||f.Risk!=100||!strings.Contains(strings.Join(f.Reasons,";"),label){t.Fatalf("IOC match failed for %s",label)}
   matches++
  }
  first:="0";if strings.HasPrefix(hash,"0"){first="1"}
  changed:=first+hash[1:]
  if _,exists:=s.IOC[changed];exists{t.Fatal("negative control collision")}
  if f:=analyze("metadata-only.txt",[]byte("Benign laboratory placeholder."),changed,s.IOC);f!=nil{t.Fatal("changed digest incorrectly matched")}
  negative++
 }
 t.Logf("Public identifiers: 12; exact/case variants: %d; altered digest controls: %d; active malware executed: 0",matches,negative)
}
func TestBenignFilesAndActualDigestPath(t *testing.T){
 s:=New();if err:=s.LoadIOC("lab/public_ioc_2026-11-25.sha256");err!=nil{t.Fatal(err)}
 root:=t.TempDir()
 var sample string
 for i:=0;i<100;i++{
  p:=filepath.Join(root,fmt.Sprintf("control-%03d.txt",i))
  if err:=os.WriteFile(p,[]byte(fmt.Sprintf("Harmless generated text control %d.",i)),0600);err!=nil{t.Fatal(err)}
  if i==0{sample=p}
 }
 report,err:=s.Scan(context.Background(),root,nil)
 if err!=nil||report.Files!=100||len(report.Findings)!=0||len(report.Errors)!=0||report.Skipped!=0{t.Fatalf("benign controls: %+v %v",report,err)}
 b,err:=os.ReadFile(sample);if err!=nil{t.Fatal(err)}
 sum:=sha256.Sum256(b);hash:=hex.EncodeToString(sum[:])
 s.IOC[hash]="LAB harmless digest control"
 report,err=s.Scan(context.Background(),sample,nil)
 if err!=nil||len(report.Findings)!=1||report.Findings[0].SHA256!=hash||report.Findings[0].Risk!=100{t.Fatalf("actual file hashing test: %+v %v",report,err)}
 t.Log("100 generated benign files: zero alerts; actual complete-file SHA-256 control: matched; no malware files")
}
