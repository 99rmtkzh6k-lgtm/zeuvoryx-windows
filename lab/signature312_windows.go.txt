package main
import (
 "context"
 "time"
 "github.com/zeuvoryx/zeuvoryx/engine"
 "github.com/zeuvoryx/zeuvoryx/signatureav"
)
// Called once per normal desktop scan, outside the State mutex, before persisting the final report.
func (s *State) signaturePhase312(ctx context.Context, rep *engine.Report, customPath string, automatic bool) {
 rep.SignatureState="pending"
 defer func(){rep.Finished=time.Now();if ctx.Err()!=nil{rep.Cancelled=true;rep.SignatureState="partial"};s.publishSignatureProgress314(rep)}()
 if ctx.Err()!=nil{rep.SignatureState="partial";return}
 tool,e:=clamExecutable()
 if e!=nil{rep.SignatureState="unavailable";rep.SignatureProblem=e.Error();return}
 roots:=rep.SignatureRoots
 if !automatic{roots=[]string{customPath}}
 if len(roots)==0{rep.SignatureState="unavailable";rep.SignatureProblem="No verified roots for signature scan";return}
 rep.SignatureState="running"
 if _,e:=ensureClam314(ctx,false);e!=nil{rep.SignatureState="unavailable";rep.SignatureProblem=e.Error();return}
 r,e:=signatureav.ScanRoots(signatureav.WithObserver(ctx,publishSignature314),tool,roots,"")
 rep.SignatureReports=append(rep.SignatureReports,r)
 if e!=nil||r.Partial{rep.SignatureState="partial"}else{rep.SignatureState="finished"}
 s.publishSignatureProgress314(rep)
}

// Publish the batch result; the live signal buffer separately supplies partial findings.
func(s *State) publishSignatureProgress314(rep *engine.Report){
 s.mu.Lock();defer s.mu.Unlock();if !s.running{return}
 s.report.SignatureState=rep.SignatureState
 s.report.SignatureProblem=rep.SignatureProblem
 s.report.SignatureRoots=append([]string(nil),rep.SignatureRoots...)
 s.report.SignatureReports=append([]signatureav.Report(nil),rep.SignatureReports...)
}
