func TestZStressScanner5M(t *testing.T){
 if os.Getenv("ZEUVORYX_QA20")!="1"{t.Skip("opt-in workload")}
 start:=time.Now();var n [3]int
 for i:=0;i<5000000;i++{
  data:=strconv.AppendInt([]byte("Harmless quality fixture "),int64(i),10)
  sum:=sha256.Sum256(data);hash:=hex.EncodeToString(sum[:])
  path:="ordinary.txt";var ioc map[string]string
  switch i%3{case 1:ioc=map[string]string{hash:"synthetic known fixture"};case 2:path="document.pdf.exe"}
  f:=analyze(path,data,hash,ioc)
  switch i%3{
  case 0:if f!=nil{t.Fatalf("benign fixture %d: %+v",i,f)}
  case 1:if f==nil||f.Evidence!="ioc_match"||!f.QuarantineEligible||f.Risk!=100{t.Fatalf("IOC fixture %d: %+v",i,f)}

  };n[i%3]++

}
func TestZRealWindowsFiles(t *testing.T){
 if os.Getenv("ZEUVORYX_QA20")!="1"||runtime.GOOS!="windows"{t.Skip("opt-in Windows fixture")}
 root:=os.Getenv("SystemRoot");if root==""{t.Fatal("SystemRoot missing")}
 s:=New();n:=0;start:=time.Now();var total int64
 for _,name:=range []string{"kernel32.dll","ntdll.dll","advapi32.dll","user32.dll","shell32.dll","wintrust.dll","crypt32.dll","notepad.exe","cmd.exe","reg.exe"}{
  p:=filepath.Join(root,"System32",name);if _,e:=os.Stat(p);os.IsNotExist(e){t.Logf("unavailable: %s",name);continue}
  r,e:=s.Scan(context.Background(),p,nil);if e!=nil||r.Files!=1{t.Fatalf("read %s: %+v %v",name,r,e)}
  n++;total+=r.Bytes
  for _,f:=range r.Findings{if f.Evidence!="heuristic"||f.QuarantineEligible{t.Fatalf("unsafe system-file finding %s: %+v",name,f)}}
  t.Logf("real OS file %s bytes=%d heuristic_alerts=%d",name,r.Bytes,len(r.Findings))
 }
 if n<5{t.Fatal("too few real system files")}
 t.Logf("real OS files=%d bytes=%d elapsed=%s",n,total,time.Since(start))
}
