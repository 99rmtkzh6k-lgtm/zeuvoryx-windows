func TestLabProtectionControls(t *testing.T){
 s:=New()
 root:=t.TempDir()
 content:=[]byte("Harmless held-out laboratory content 2026-11-25.")
 sum:=sha256.Sum256(content)
 hash:=hex.EncodeToString(sum[:])
 s.IOC[hash]="LAB harmless signature control"
 plain:=filepath.Join(root,"control.txt")
 if err:=os.WriteFile(plain,content,0600);err!=nil{t.Fatal(err)}
 arc:=filepath.Join(root,"control.zip")
 out,err:=os.Create(arc);if err!=nil{t.Fatal(err)}
 z:=zip.NewWriter(out);entry,err:=z.Create("entry.txt");if err!=nil{t.Fatal(err)}
 if _,err=entry.Write(content);err!=nil{t.Fatal(err)}
 if err=z.Close();err!=nil{t.Fatal(err)};if err=out.Close();err!=nil{t.Fatal(err)}
 report,err:=s.Scan(context.Background(),root,nil)
 if err!=nil||report.Files!=2||report.Skipped!=0||len(report.Findings)!=2{t.Fatalf("coverage: %+v %v",report,err)}
 archived,normal:=false,false
 for _,f:=range report.Findings{
  if f.SHA256!=hash||f.Risk!=100{t.Fatal(f)}
  if strings.Contains(f.Path," :: "){archived=true}else{normal=true}
 }
 if !archived||!normal{t.Fatal("missing plain or archive match")}
 ctx,cancel:=context.WithCancel(context.Background());cancel()
 stopped,err:=s.Scan(ctx,root,nil)
 if err!=nil||!stopped.Cancelled||stopped.Files!=0{t.Fatalf("cancel state %+v %v",stopped,err)}
 b,err:=json.Marshal(report);if err!=nil{t.Fatal(err)}
 var round Report;if err=json.Unmarshal(b,&round);err!=nil||round.Files!=2||len(round.Findings)!=2{t.Fatal("report not reproducible",err)}
 t.Log("Protection proxies: plain and ZIP digest controls matched; cancelled scan and JSON verified. No live malware.")
}

func TestLabBenignAlertSurvey(t *testing.T){
 s:=New()
 root:=t.TempDir()
 // Deliberately benign content; documentation may contain security terminology.
 types:=[]struct{ext,body string}{
  {".txt","Ordinary notes for a family meeting."},
  {".csv","product,quantity,price\nbook,1,10"},
  {".json","{\"enabled\":true,\"name\":\"sample\"}"},
  {".ps1","# Harmless PowerShell documentation\nWrite-Output 'hello'"},
  {".js","// Harmless example\nconst value = 1;"},
  {".md","Glossary: Invoke-Expression and FromBase64String( are terms in PowerShell documentation."},
 }
 for i:=0;i<600;i++{
  x:=types[i%len(types)];p:=filepath.Join(root,fmt.Sprintf("benign-%04d%s",i,x.ext))
  if err:=os.WriteFile(p,[]byte(fmt.Sprintf("%s\n",x.body)),0600);err!=nil{t.Fatal(err)}
 }
 r,err:=s.Scan(context.Background(),root,nil)
 if err!=nil||r.Files!=600||r.Skipped!=0||len(r.Errors)!=0{t.Fatalf("incomplete survey %+v %v",r,err)}
 byType:=map[string]int{}
 for _,f:=range r.Findings{byType[filepath.Ext(f.Path)]++}
 t.Logf("BENIGN_SURVEY files=%d advisory_alerts=%d by_extension=%v; no blocking measured",r.Files,len(r.Findings),byType)
 // Diagnostic survey, not a claim of acceptable false-alert rate.
}

  n,e:=io.Copy(sha256.New(),in);ce:=in.Close();if e!=nil||ce!=nil{t.Fatal(e,ce)};bytesRead+=n
